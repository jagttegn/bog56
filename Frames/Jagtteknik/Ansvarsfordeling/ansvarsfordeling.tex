\section{Ansvarsfordeling under jagt}

\begin{frame}
  \vspace{3em}
  \begin{center}
    {\LARGE \textbf{Ansvarsfordeling}
    }
    \\
    \vspace{1em}
  \end{center}
\end{frame}

\begin{frame}{Jagtlederens ansvar}
    \begin{itemize}
        \item Afholder klar parole (gennemgår sikkerhed, arter, struktur)
        \item Gennemførelse (placering, farlige områder, start, slut)
        \item Fordeling af andet ansvar
        \item Instruktér alle om poster og naboposter
        \item Alle tager hovedbeklædning af
    \end{itemize}
\end{frame}

\begin{frame}{Jægerens ansvar}
    \begin{itemize}
        \item Følg parole og jagtlederens instruktioner
        \item Sikkerhedsvinkler
        \item Korrekt udstyr til formålet
        \item Forsvarligt kuglefang
        \item Kun etiske skud til tilladt vildt
        \item Måske synligt tøj (orange) + styr på udstyr
        \item Gå bag jagtlederen, sig tak for post/ løft hovedbeklædning
    \end{itemize}
\end{frame}
